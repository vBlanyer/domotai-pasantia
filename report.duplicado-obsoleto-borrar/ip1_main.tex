% El entorno thebibliography encabeza con \chapter*, que activa el estilo de
% página «plain» y dejaría Referencias sin número. Como la norma 2.6 excluye a
% Referencias de la numeración de capítulos, se redefine para que use el mismo
% encabezado que las demás secciones sin numerar.
\makeatletter
\renewenvironment{thebibliography}[1]{%
  \seccionSinNumerar{\bibname}%
  \list{\@biblabel{\@arabic\c@enumiv}}%
       {\settowidth\labelwidth{\@biblabel{#1}}%
        \leftmargin\labelwidth \advance\leftmargin\labelsep
        \usecounter{enumiv}\let\p@enumiv\@empty
        \renewcommand\theenumiv{\@arabic\c@enumiv}}%
  \sloppy\clubpenalty4000\@clubpenalty\clubpenalty
  \widowpenalty4000\sfcode`\.\@m}
  {\def\@noitemerr{\@latex@warning{Empty `thebibliography' environment}}%
   \endlist}
\makeatother
